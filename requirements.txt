# ---------------------------------------------------------------------------
# Net-Peekier dependencies
#
# Only psutil is required. Everything below it is optional and the app degrades
# gracefully when it's missing (features just show "--" / "recorded only" /
# psutil-only mode). Install the optional extras for the full experience.
# ---------------------------------------------------------------------------

# Required -------------------------------------------------------------------
psutil>=5.9

# Optional: per-process speeds, packet capture, blocking & throttling --------
# Enables real per-process upload/download speeds, the live packet + hex view,
# and enforced blocking/throttling. Needs the WinDivert driver, which pydivert
# bundles. Windows only; run as Administrator.
#     pip install pydivert
pydivert>=2.1 ; platform_system == "Windows"

# Optional: system-stats temperatures (CPU / GPU / RAM) ----------------------
# The dashboard's System panel gets CPU/RAM load + clock from psutil with no
# extra install. TEMPERATURES on Windows need LibreHardwareMonitor via this
# thin pythonnet layer (pulls in pythonnet; needs .NET and Administrator).
# Without it, temperature fields simply show "--".
#     pip install HardwareMonitor
#
# NOTE: many motherboard/CPU sensors also require the PawnIO kernel driver.
# It is NOT a pip package -- install it once via winget (then reboot if asked):
#     winget install PawnIO
# Without PawnIO, CPU package temperature often reads as unavailable.
HardwareMonitor>=0.9 ; platform_system == "Windows"

# Optional: NVIDIA GPU fallback ----------------------------------------------
# If HardwareMonitor isn't available, NVIDIA GPUs can still report temp/clock/
# load through NVML. Skip if you use HardwareMonitor above or have no NVIDIA GPU.
#     pip install nvidia-ml-py
nvidia-ml-py>=12.0

# Optional: extra image formats in the About > Readme viewer -----------------
# The built-in README viewer shows PNG/GIF images with no extra install. Add
# Pillow only if your README embeds JPG/WebP images and you want them rendered;
# without it those images fall back to their alt-text.
#     pip install Pillow
Pillow>=9.0
